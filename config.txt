# --- Piloto automatico ----------------------------------------------------
# Busca sozinho nas lojas, escolhe as melhores ofertas e publica no WhatsApp e
# no Telegram com o seu link de afiliado. Estas linhas sao relidas a cada
# ciclo: da para mudar sem reiniciar. Painel e historico: no portal.
# 1 = ligado. 0 = desligado (o portal continua funcionando normal).
PILOTO_ATIVO=1
# Intervalo entre ciclos, em minutos (sorteado entre o minimo e o maximo).
PILOTO_INTERVALO_MIN=10
PILOTO_INTERVALO_MAX=20
# Lojas buscadas. Shopee fica de fora ate a API de afiliados ser liberada.
PILOTO_LOJAS=mercadolivre,amazon,kabum,aliexpress
# Prioridade na hora de escolher o que postar: entre as ofertas que passaram
# em todas as regras, sai primeiro a da loja que vem antes nesta lista (a nota
# so desempata dentro da mesma loja). Loja fora da lista fica por ultimo.
# Com PILOTO_MAX_POR_LOJA_NO_CICLO=1, cada ciclo posta a melhor oferta de cada
# loja, na ordem: primeiro Mercado Livre, depois Amazon, e assim por diante.
PILOTO_PRIORIDADE_LOJAS=mercadolivre,amazon,kabum,aliexpress
# Categorias em rodizio: "todas" ou chaves separadas por virgula
# (tecnologia,notebooks,smartphones,hardware,perifericos,games,audio,
#  smart_home,wearables,tablets,tv_video).
PILOTO_CATEGORIAS=todas
# Quantas categorias cada ciclo cobre (1 categoria leva de 3 a 6 minutos).
PILOTO_CATEGORIAS_POR_CICLO=1
# Produtos lidos por loja em cada termo de busca.
PILOTO_ITENS_POR_LOJA=8
# Limites de publicacao (WhatsApp bane numero que dispara demais).
PILOTO_POSTS_POR_CICLO=2
PILOTO_MAX_POR_HORA=8
PILOTO_MAX_POR_LOJA_NO_CICLO=2
# Desconto minimo (%) e preco minimo (R$) para virar postagem.
PILOTO_DESCONTO_MIN=15
PILOTO_PRECO_MIN=15
# Horario sem postagens (continua coletando precos). Ex.: 23-7. Vazio = nunca.
PILOTO_SILENCIO=23-7
# O mesmo produto so volta depois de X dias, a nao ser que o preco caia mais
# Y% em relacao a ultima postagem.
PILOTO_REPOSTAR_DIAS=3
PILOTO_QUEDA_PARA_REPOSTAR=5
# Preco de mercado: cada busca traz o mesmo produto de varias lojas. Oferta
# acima de X vezes o preco tipico das OUTRAS lojas (e fora da curva) e'
# descartada, e preco "de" acima desse teto e' tratado como inventado (o post
# sai sem o "De:" e com o desconto real). Ex.: 1.3 = ate 30% acima do tipico.
PILOTO_TETO_MERCADO=1.3
# 1 = nao posta produto usado, seminovo, recondicionado, vitrine ou open box.
PILOTO_BLOQUEAR_USADOS=1
# 1 = aprendizado de precos ligado: com 3+ dias de historico o piloto usa o
#     preco normal que ele mesmo observou (um preco por dia), pega o "sobe
#     antes, da desconto depois" e aprende o quanto o de/por de cada loja
#     costuma ser real. Detalhes no LEIA-ME.txt. 0 = regra antiga.
PILOTO_APRENDIZADO=1
# 1 = quando o piloto ve de novo um produto que ja postou, avisa o site do
#     preco atual: o catalogo acompanha a loja e a oferta que some sai do ar
#     sozinha. Precisa de SITE_URL e SITE_TOKEN.
PILOTO_ATUALIZAR_SITE=1

PILOTO_CUPONS=1

# Paginas (de 100 produtos) lidas da lista de cada cupom.
